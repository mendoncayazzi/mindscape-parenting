  <!-- JAVASCRIPT LOGIC -->

  <script>

    function switchTab(tabName) {

      ['journal', 'lectures', 'play', 'reset', 'partner'].forEach(t => {

        document.getElementById(`tab-${t}`).classList.add('hide');

        document.getElementById(`nav-${t}`).classList.remove('text-indigo-600');

        document.getElementById(`nav-${t}`).classList.add('text-slate-400');

      });

      document.getElementById(`tab-${tabName}`).classList.remove('hide');

      document.getElementById(`nav-${tabName}`).classList.add('text-indigo-600');

      document.getElementById(`nav-${tabName}`).classList.remove('text-slate-400');

    }


    function openSubscription() {

      window.location.href = STRIPE_CHECKOUT_URL;

    }


    async function saveJournal(e) {

      e.preventDefault();

      const entry = {

        email: document.getElementById('j-email').value,

        date: document.getElementById('j-date').value,

        mood: document.getElementById('j-mood').value,

        screen_time: document.getElementById('j-screen').value,

        play_time: document.getElementById('j-play').value,

        interest: document.getElementById('j-interest').value,

        tantrum: document.getElementById('j-tantrum').value,

        repair: document.getElementById('j-repair').value

      };


      if (supabaseClient) {

        const { error } = await supabaseClient.from('journal_entries').insert([entry]);

        if (error) {

          alert('Error saving report: ' + error.message);

          return;

        }

      }

      alert('Daily report successfully saved!');

      document.getElementById('journal-form').reset();

    }

  </script>
